cmake_minimum_required(VERSION 4.3.3)
set(CMAKE_CXX_STANDARD 26)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(CMAKE_VERSION VERSION_LESS 4.4)
  set(CMAKE_EXPERIMENTAL_CXX_IMPORT_STD "451f2fe2-a8a2-47c3-bc32-94786d8fc91b")
else()
  set(CMAKE_EXPERIMENTAL_CXX_IMPORT_STD "f35a9ac6-8463-4d38-8eec-5d6008153e7d")
endif()
set(CMAKE_CXX_MODULE_STD ON)
project(AnnoTestConsumer LANGUAGES CXX)
if(CMAKE_VERSION VERSION_LESS 4.4)
  string(APPEND CMAKE_CXX_FLAGS " -freflection")
endif()

enable_testing()
if(ANNOTEST_SOURCE_DIR)
  set(BUILD_TESTS OFF)
  include(FetchContent)
  FetchContent_Declare(annotest SOURCE_DIR "${ANNOTEST_SOURCE_DIR}")
  FetchContent_MakeAvailable(annotest)
  if(TARGET example1_basic OR TARGET AssertionTests)
    message(FATAL_ERROR "Embedding AnnoTest must not add examples or self-tests by default")
  endif()
else()
  find_package(AnnoTest CONFIG REQUIRED)
  # Including the package twice must be harmless.
  find_package(AnnoTest CONFIG REQUIRED)
endif()

add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE annotest::annotest)
target_compile_options(consumer PRIVATE -fcontracts -fcontract-evaluation-semantic=observe)
annotest_discover_tests(consumer)
